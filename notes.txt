# domain/entities.py

VALID_TRANSITIONS = {
    RunStatus.PENDING:           {RunStatus.RUNNING},
    RunStatus.RUNNING:           {RunStatus.OUTPUTS_GENERATED, RunStatus.FAILED},
    RunStatus.OUTPUTS_GENERATED: {RunStatus.COMPLETED, RunStatus.CHECKS_FAILED, RunStatus.CHECKS_NOT_EXECUTED},
    RunStatus.COMPLETED:         set(),  # terminale — nessuna transizione possibile
    RunStatus.CHECKS_FAILED:     set(),  # terminale
    RunStatus.CHECKS_NOT_EXECUTED: set(), # terminale
    RunStatus.FAILED:            set(),  # terminale
}

@dataclass
class ModelRun:

    def _transition_to(self, new_status: RunStatus) -> None:
        allowed = VALID_TRANSITIONS.get(self.status, set())
        if new_status not in allowed:
            raise InvalidStateTransitionError(
                f"Transizione '{self.status}' → '{new_status}' non consentita"
            )
        self.status = new_status
        self.status_log.append(RunStatusLog(
            run_id=self.id,
            status=new_status,
            timestamp=datetime.now(timezone.utc),
        ))

    def mark_running(self) -> None:
        self._transition_to(RunStatus.RUNNING)

    def mark_outputs_generated(self, outputs: dict) -> None:
        self._transition_to(RunStatus.OUTPUTS_GENERATED)
        self.outputs = outputs

    def mark_completed(self, check_results: list[CheckResult]) -> None:
        self._transition_to(RunStatus.COMPLETED)
        self.completed_at = datetime.now(timezone.utc)
        self.check_results = check_results

    def mark_checks_failed(self, check_results: list[CheckResult]) -> None:
        self._transition_to(RunStatus.CHECKS_FAILED)
        self.completed_at = datetime.now(timezone.utc)
        self.check_results = check_results

    def mark_checks_not_executed(self, reason: str) -> None:
        self._transition_to(RunStatus.CHECKS_NOT_EXECUTED)
        self.completed_at = datetime.now(timezone.utc)
        self.error_message = reason

    def mark_failed(self, error: str) -> None:
        self._transition_to(RunStatus.FAILED)
        self.completed_at = datetime.now(timezone.utc)
        self.error_message = error








Append-only si implementa su tre livelli — ognuno aggiunge un layer di protezione.

---

## Livello 1 — Domain: nessun metodo di modifica

Le entity non hanno setter o metodi che modificano i campi. L'unica operazione consentita è `create()`:

```python
@dataclass(frozen=True)  # frozen=True — fisicamente immutabile in Python
class ModelVersion:
    model_id: str
    version: str
    code_version: str
    approved: bool = False
    created_at: datetime = field(default_factory=lambda: datetime.now(timezone.utc))

    @classmethod
    def create(cls, model_id: str, version: str, code_version: str) -> "ModelVersion":
        return cls(
            model_id=model_id,
            version=version,
            code_version=code_version,
        )
```

`frozen=True` qui ha senso — a differenza di `ModelRun` che cambia stato, `ModelVersion` non deve mai cambiare dopo la creazione.

---

## Livello 2 — Repository: nessun metodo update

L'interfaccia espone solo `save()` per INSERT e `get()` per SELECT — nessun `update()`:

```python
# application/interfaces/repository_interfaces.py

class IModelVersionRepository(ABC):

    @abstractmethod
    def save(self, version: ModelVersion) -> ModelVersion:
        """Solo INSERT — mai UPDATE"""
        pass

    @abstractmethod
    def get(self, model_id: str, version: str) -> ModelVersion:
        pass

    @abstractmethod
    def exists(self, model_id: str, version: str) -> bool:
        pass
```

E l'implementazione Django raisa esplicitamente se qualcuno prova a fare update:

```python
class DjangoModelVersionRepository(IModelVersionRepository):

    def save(self, version: ModelVersion) -> ModelVersion:
        # controlla che non esista già — se esiste, blocca
        if ModelVersionORM.objects.filter(
            model_id=version.model_id,
            model_version=version.version
        ).exists():
            raise VersionAlreadyExistsError(
                f"Versione '{version.version}' già esistente per modello '{version.model_id}'"
            )

        orm_obj = ModelVersionORM(
            model_id_id=version.model_id,
            model_version=version.version,
            code_version=version.code_version,
            approved=version.approved,
        )
        orm_obj.save()
        return version
```

## Livello 3 — Database: constraint e trigger PostgreSQL

Questo è il livello più blindato — anche se qualcuno bypassa Python e va direttamente sul DB, non può modificare niente.

Si fa con una **Django migration custom** che aggiunge un trigger PostgreSQL:

```python
# infrastructure/django/migrations/0002_append_only_trigger.py

from django.db import migrations

class Migration(migrations.Migration):

    dependencies = [("django", "0001_initial")]

    operations = [
        migrations.RunSQL(
            sql="""
                CREATE OR REPLACE FUNCTION prevent_update_delete()
                RETURNS TRIGGER AS $$
                BEGIN
                    RAISE EXCEPTION 'Tabella append-only: UPDATE e DELETE non consentiti';
                END;
                $$ LANGUAGE plpgsql;

                CREATE TRIGGER no_update_model_versions
                    BEFORE UPDATE OR DELETE ON model_versions
                    FOR EACH ROW EXECUTE FUNCTION prevent_update_delete();

                CREATE TRIGGER no_update_parameter_versions
                    BEFORE UPDATE OR DELETE ON model_parameters
                    FOR EACH ROW EXECUTE FUNCTION prevent_update_delete();
            """,
            reverse_sql="""
                DROP TRIGGER IF EXISTS no_update_model_versions ON model_versions;
                DROP TRIGGER IF EXISTS no_update_parameter_versions ON model_parameters;
                DROP FUNCTION IF EXISTS prevent_update_delete();
            """
        )
    ]
```

## I tre livelli insieme

```
frozen dataclass         → impossibile modificare in Python
repository senza update  → impossibile modificare via codice applicativo  
trigger PostgreSQL        → impossibile modificare anche via SQL diretto
```

Chiunque provi a modificare una riga — dal codice, da una migration, da psql direttamente — riceve un errore esplicito. Questo è append-only blindato.
